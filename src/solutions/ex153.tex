\begin{exercise}{153}
Show that a fully faithful and essentially surjective functor is invertible.

\begin{proof}
Let $f\colon\mathcal{C}\to\mathcal{D}$ be such a functor.
Theorem~2.3.20 makes $f$ a Joyal equivalence.
By Definition~2.1.14 and Remark~2.1.15, there is a functor
$g\colon\mathcal{D}\to\mathcal{C}$ with natural equivalences
\[
 gf\simeq\mathrm{id}_{\mathcal{C}},\qquad
 fg\simeq\mathrm{id}_{\mathcal{D}}.
\]
Thus $g$ is an inverse to $f$ in $\mathrm{Cat}_\infty$.
\end{proof}
\end{exercise}
